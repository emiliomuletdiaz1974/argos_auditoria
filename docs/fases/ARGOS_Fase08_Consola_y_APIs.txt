0. La capa C7 en conjunto
La consola es el rostro del producto y su regla de diseño viene del posicionamiento: ARGOS no es un panel de gráficas, es un instrumento de trabajo del DPO y del CISO — cada vista responde una pregunta de su jornada (¿qué hay?, ¿qué está pasando?, ¿qué hago primero?, ¿qué enseño?) y cada acción sensible deja rastro en el diario. La API que la sirve es la misma que se ofrece al cliente para integración: no hay endpoints privados de la consola, porque un producto de evidencia no puede tener dos verdades.
La arquitectura de la capa: FastAPI como API única (ARG-071) con autorización por rol y doble aprobación como middleware (ARG-072), sesión OIDC contra el Keycloak del appliance con PKCE (ARG-073); frontend React servido como estáticos desde el propio appliance (sin CDN: P-03), con las cuatro vistas (ARG-074..077) y el chat (ARG-078) montados sobre el sistema de diseño (ARG-080); y los webhooks firmados (ARG-079) que llevan hallazgos y campañas al ITSM del cliente — porque el trabajo de subsanación vive donde el cliente trabaja, no donde ARGOS quiera.

ARG-071 · La API REST única
ANÁLISIS
Los recursos de la API calcan el dominio: systems, inventory (catálogo+grafo vía ARG-029), campaigns (con sus gates y su progreso en vivo), findings (con transiciones), evidence (índice y descargas), credentials, assistant. Las decisiones transversales: paginación por cursor opaco en todo listado (offset se degrada con volumen), errores RFC 9457 (problem+json: tipo, título, detalle, instancia — depurables sin leer código), idempotencia por clave en todo POST que crea (el cliente ITSM reintenta sin duplicar campañas) y versionado por cabecera con v1 congelada al GA. El contrato OpenAPI se genera del código y se publica en el propio appliance: la documentación no puede divergir.
DISEÑO
Aplicación FastAPI con routers por recurso, dependencias comunes (auth, paginación, idempotencia) y el middleware de auditoría: toda mutación escribe en el diario con actor, recurso y resumen — la consola hereda la trazabilidad sin que cada endpoint la recuerde. Se muestra el núcleo: aplicación, dependencias y el router de campañas como patrón (los demás lo repiten).

"""API única de ARGOS: la misma para consola e integración."""
import base64, json, uuid
from fastapi import FastAPI, Depends, Request, HTTPException
from fastapi.responses import JSONResponse
from argos_auth.fastapi import current_user
from argos_db import pool
 
app = FastAPI(title="ARGOS API", version="1.0",
              docs_url="/api/docs", openapi_url="/api/openapi.json")
 
# ---------- errores problem+json (RFC 9457) ----------
@app.exception_handler(HTTPException)
async def problem_handler(req: Request, exc: HTTPException):
    return JSONResponse(status_code=exc.status_code, content={
        "type": f"https://argos.local/errors/{exc.status_code}",
        "title": exc.detail if isinstance(exc.detail, str) else "error",
        "status": exc.status_code,
        "instance": str(req.url.path)},
        media_type="application/problem+json")
 
# ---------- paginación por cursor opaco ----------
def pagination(cursor: str | None = None, limit: int = 50):
    after = 0
    if cursor:
        after = int(base64.urlsafe_b64decode(cursor).decode())
    return {"after": after, "limit": min(limit, 200)}
 
def next_cursor(last_id: int) -> str:
    return base64.urlsafe_b64encode(str(last_id).encode()).decode()
 
# ---------- idempotencia en POST creadores ----------
async def idempotency(request: Request):
    key = request.headers.get("Idempotency-Key")
    if request.method == "POST" and key:
        with pool.connection() as db:
            row = db.execute("""SELECT response FROM argos.idempotency
                WHERE key=%s AND path=%s""",
                (key, request.url.path)).fetchone()
            if row:
                raise HTTPException(200, detail=row[0])   # respuesta cacheada
    return key
 
# ---------- auditoría de mutaciones ----------
@app.middleware("http")
async def audit_mutations(request: Request, call_next):
    resp = await call_next(request)
    if request.method in ("POST", "PATCH", "DELETE") \
       and resp.status_code < 400:
        user = getattr(request.state, "user", {"sub": "anonimo"})
        with pool.connection() as db:
            db.execute("SELECT argos.journal_append(%s,%s,%s::jsonb)",
                (f"user:{user['sub']}", "api.mutation",
                 json.dumps({"method": request.method,
                             "path": request.url.path,
                             "status": resp.status_code})))
            db.commit()
    return resp
 
from routes import systems, inventory, campaigns, findings, \
                   evidence, credentials, assistant, approvals
for r in (systems, inventory, campaigns, findings,
          evidence, credentials, assistant, approvals):
    app.include_router(r.router, prefix="/api/v1")

"""Campañas: crear (idempotente), listar, progreso en vivo, compilación previa."""
import json, uuid
from fastapi import APIRouter, Depends, HTTPException
from pydantic import BaseModel
from temporalio.client import Client
from argos_auth.fastapi import require_role
from app import pagination, next_cursor, idempotency
from argos_db import pool
 
router = APIRouter(tags=["campaigns"])
 
class NewCampaign(BaseModel):
    name: str
    scope: dict                 # {system_ids: [...]} o {vertical: 'sanidad'}
    ontology_pin: str | None = None
 
@router.post("/campaigns",
             dependencies=[Depends(require_role("gestor_campanas"))])
async def create(body: NewCampaign, idem=Depends(idempotency)):
    cid = str(uuid.uuid7())
    with pool.connection() as db:
        db.execute("""INSERT INTO argos.campaigns
            (id, name, scope, status, created_at)
            VALUES (%s,%s,%s::jsonb,'planificada', now())""",
            (cid, body.name, json.dumps(body.scope)))
        if idem:
            db.execute("""INSERT INTO argos.idempotency (key, path, response)
                VALUES (%s,'/api/v1/campaigns',%s::jsonb)""",
                (idem, json.dumps({"id": cid})))
        db.commit()
    t = await Client.connect("temporal.argos-core:7233")
    await t.start_workflow("CampaignWorkflow", cid,
                           id=f"campaign-{cid}",
                           task_queue="argos-challenge")
    return {"id": cid, "status": "planificada"}
 
@router.get("/campaigns")
async def list_campaigns(page=Depends(pagination)):
    with pool.connection() as db:
        rows = db.execute("""SELECT seq, id, name, status,
                created_at::text FROM argos.campaigns
            WHERE seq > %s ORDER BY seq LIMIT %s""",
            (page["after"], page["limit"])).fetchall()
    items = [dict(zip(("seq","id","name","status","created_at"), r))
             for r in rows]
    return {"items": items,
            "next": next_cursor(items[-1]["seq"]) if items else None}
 
@router.get("/campaigns/{cid}/progress")
async def progress(cid: str):
    t = await Client.connect("temporal.argos-core:7233")
    h = t.get_workflow_handle(f"campaign-{cid}")
    try:
        return await h.query("progress")     # la query de ARG-043
    except Exception:
        raise HTTPException(404, "campaña sin ejecución activa")
 
@router.get("/campaigns/{cid}/plan")
async def plan_preview(cid: str):
    """El plan compilado que el DPO lee antes de aprobar el arranque."""
    with pool.connection() as db:
        row = db.execute("""SELECT plan, ontology_version, nodes_total
            FROM argos.applicability_runs WHERE campaign_id=%s
            ORDER BY created_at DESC LIMIT 1""", (cid,)).fetchone()
    if not row:
        raise HTTPException(404, "plan no compilado todavía")
    return {"ontology": row[1], "nodes": row[2], "plan": row[0]}
 

ARG-072 · Autorización: roles, ámbitos y doble control
ANÁLISIS
Los cuatro roles de la fase 1 se convierten aquí en una matriz explícita permiso→rol que vive en un fichero versionado — no dispersa en decoradores—: la matriz es revisable por el cliente en la implantación (¿quién puede aceptar riesgo? es una pregunta de gobierno, no de código) y el middleware la aplica. El token JWT del Keycloak trae los roles; el middleware resuelve y deja el usuario en request.state para la auditoría de ARG-071.

services/api/authz/enforce.py
"""Autorización por matriz versionada; separación de deberes nominal."""
import pathlib, yaml
from fastapi import Depends, HTTPException, Request
from argos_auth.fastapi import current_user
 
MATRIX = yaml.safe_load(pathlib.Path(
    "/opt/argos/config/permissions.yaml").read_text())
 
def require_perm(perm: str):
    allowed = set(MATRIX.get(perm, []))
    if not allowed:
        raise RuntimeError(f"permiso no declarado en la matriz: {perm}")
    async def dep(request: Request, user=Depends(current_user)):
        roles = set(user.get("roles", []))
        if not roles & allowed:
            raise HTTPException(403,
                f"requiere uno de: {sorted(allowed)}")
        request.state.user = user
        return user
    return dep
 

ARG-073 · Sesión OIDC del frontend
ANÁLISIS
La consola es una SPA servida por el propio appliance y su sesión sigue el patrón recomendado para SPAs sin secretos: Authorization Code con PKCE contra el Keycloak local, tokens en memoria (jamás localStorage: XSS no debe encontrar nada persistente), renovación silenciosa por refresh token en cookie HttpOnly+Secure+SameSite=Strict que solo ve el endpoint de renovación, y expiración de sesión alineada con la política del cliente (importable de su IdP corporativo si el Keycloak federa, que es el caso habitual en AAPP).
DISEÑO
Del lado del frontend, un módulo auth.js minimalista (sin dependencia pesada): login() construye la URL de autorización con el verifier PKCE, callback() canjea el código, y fetchApi() adjunta el Bearer y reintenta una vez tras renovar. Del lado API, /auth/refresh custodia la cookie. Se muestra el módulo frontend, que es donde están las decisiones.

console/src/lib/auth.js
// Sesión OIDC con PKCE. Tokens SOLO en memoria; refresh en cookie HttpOnly.
const KC = "/auth/realms/argos/protocol/openid-connect";
const CLIENT = "argos-console";
let accessToken = null;              // memoria: se pierde al recargar (ok)
 
const b64url = (buf) =>
  btoa(String.fromCharCode(...new Uint8Array(buf)))
    .replace(/\+/g, "-").replace(/\//g, "_").replace(/=+$/, "");
 
async function pkce() {
  const verifier = b64url(crypto.getRandomValues(new Uint8Array(32)));
  const digest = await crypto.subtle.digest("SHA-256",
    new TextEncoder().encode(verifier));
  sessionStorage.setItem("pkce_v", verifier);   // solo el verifier, efímero
  return { verifier, challenge: b64url(digest) };
}
 
export async function login() {
  const { challenge } = await pkce();
  const p = new URLSearchParams({
    client_id: CLIENT, response_type: "code",
    redirect_uri: location.origin + "/callback",
    scope: "openid profile roles",
    code_challenge: challenge, code_challenge_method: "S256",
  });
  location.assign(`${KC}/auth?${p}`);
}
 
export async function callback() {
  const code = new URLSearchParams(location.search).get("code");
  const body = new URLSearchParams({
    grant_type: "authorization_code", client_id: CLIENT,
    code, redirect_uri: location.origin + "/callback",
    code_verifier: sessionStorage.getItem("pkce_v"),
  });
  const r = await fetch(`${KC}/token`, { method: "POST", body,
    credentials: "include" });          // set-cookie del refresh (HttpOnly)
  const tok = await r.json();
  accessToken = tok.access_token;
  sessionStorage.removeItem("pkce_v");
  location.replace("/");
}
 
async function refresh() {
  const r = await fetch("/api/v1/auth/refresh",
    { method: "POST", credentials: "include" });
  if (!r.ok) { login(); return false; }
  accessToken = (await r.json()).access_token;
  return true;
}
 
export async function fetchApi(path, opts = {}) {
  const go = () => fetch(path, { ...opts, headers: {
    ...opts.headers, Authorization: `Bearer ${accessToken}`,
    "Content-Type": "application/json" } });
  let r = await go();
  if (r.status === 401 && await refresh()) r = await go();
  if (r.status === 403) throw new Error("permiso insuficiente");
  return r;
}

import { useState } from "react";
import useSWR from "swr";
import { fetchApi } from "../../lib/auth";
 
const CATS = ["dato_personal", "categoria_especial.salud",
  "categoria_especial.otro", "identificador_oficial", "dato_financiero",
  "dato_contacto", "dato_localizacion", "credencial_tecnica",
  "sin_dato_personal"];

export default function ReviewQueue() {
  const { data, mutate } = useSWR(
    "/api/v1/inventory/review-queue", fetcher);
  const [busy, setBusy] = useState(null);
 
  async function resolve(item, category) {
    setBusy(item.node_key);
    await fetchApi(`/api/v1/inventory/review-queue/${item.node_key}`, {
      method: "POST",
      body: JSON.stringify({ resolution: category }),
    });

    setBusy(null);
  }
 
  if (!data) return <div className="skeleton" />;
  return (
    <section className="panel">
      <header className="panel-head">
        <h2>Cola de revisión</h2>
        <span className="badge">{data.items.length} pendientes</span>
      </header>
      {data.items.map((it) => (
        <article key={it.node_key} className="review-row">
          <div className="review-meta">
            <code>{it.table_name}.{it.column_name}</code>
            <span className="muted">{it.system_name}</span>
          </div>
          <div className="review-proposal">
            <span className="chip chip-ai"
                  title={`confianza ${it.confidence}`}>
              {it.proposed_category}
            </span>
            <small className="muted">{it.reason}</small>
          </div>
          <div className="review-actions">
            <button className="btn-primary"
                    disabled={busy === it.node_key}
                    onClick={() => resolve(it, it.proposed_category)}>
              Confirmar
            </button>
            <select defaultValue=""
                    onChange={(e) => resolve(it, e.target.value)}>
              <option value="" disabled>Corregir a…</option>
              {CATS.map((c) => <option key={c}>{c}</option>)}
            </select>
          </div>
        </article>
      ))}
    </section>
  );
}

import { useRef, useState } from "react";

function Answer({ msg }) {
  // convierte [n] en referencias desplegables
  const parts = msg.answer.split(/(\[\d+\])/g);
  return (
    <div className="msg msg-assistant">
      <p>{parts.map((p, i) => {
        const m = p.match(/^\[(\d+)\]$/);
        if (!m) return p;
        const cita = msg.citas?.find((c) => c.n === +m[1]);
        return cita ? (
          <details key={i} className="cite">
            <summary>[{m[1]}]</summary>
            <blockquote>
              <cite>{cita.ref_completa}</cite>
              <p>{cita.fragmento}</p>
            </blockquote>
          </details>) : p;
      })}</p>
      {msg.suficiente === false && (
        <div className="callout callout-warn">
          Sin base suficiente en el corpus para responder con rigor.
        </div>)}
      {msg.tools_used?.length > 0 && (
        <small className="muted">
          consultado: {[...new Set(msg.tools_used)].join(", ")}
        </small>)}
    </div>);
}
 
export default function Chat() {
  const [msgs, setMsgs] = useState([]);
  const [busy, setBusy] = useState(false);
  const input = useRef();
 
  async function send() {
    const q = input.current.value.trim();
    if (!q || busy) return;
    input.current.value = "";
    setMsgs((m) => [...m, { role: "user", answer: q }]);
    setBusy(true);
    const r = await fetchApi("/api/v1/assistant/ask", {
      method: "POST", body: JSON.stringify({ question: q }) });
    setMsgs((m) => [...m, { role: "assistant", ...(await r.json()) }]);
    setBusy(false);
  }
 
  return (
    <section className="chat">
      <div className="chat-scroll">
        {msgs.map((m, i) => m.role === "user"
          ? <div key={i} className="msg msg-user">{m.answer}</div>
          : <Answer key={i} msg={m} />)}
        {busy && <div className="msg msg-assistant skeleton-line" />}
      </div>
      <footer className="chat-input">
        <input ref={input} placeholder="Pregunta sobre normativa o estado…"
               onKeyDown={(e) => e.key === "Enter" && send()} />
        <button className="btn-primary" onClick={send}>Enviar</button>
      </footer>
      <small className="muted chat-disclaimer">
        Asistente de IA: las respuestas no constituyen veredictos de
        conformidad.
      </small>
    </section>);
}
 

ARG-079 · Webhooks firmados e integración ITSM
ANÁLISIS
La subsanación vive en el ITSM del cliente, así que ARGOS empuja: hallazgo abierto → incidencia; campaña cerrada → notificación; compuerta pendiente → aviso al aprobador. Los requisitos de un webhook serio: firma HMAC del cuerpo con secreto por suscripción (el receptor verifica que es ARGOS y que nada cambió), timestamp contra replay, reintentos con backoff exponencial y una bandeja de entregas con el estado de cada intento — porque «¿llegó al ServiceNow?» debe responderse desde la consola. Las plantillas de transformación por destino (ServiceNow, Jira, genérico) son configuración, no código.
DESARROLLO
services/api/webhooks/dispatch.py
"""Webhooks: HMAC + timestamp, reintentos con backoff, bandeja de entregas."""
import hashlib, hmac, json, time
import httpx
from temporalio import activity
from argos_db import pool
 
MAX_ATTEMPTS = 8          # backoff: 1,2,4... ~2h en total
 
def _sign(secret: bytes, ts: str, body: bytes) -> str:
    mac = hmac.new(secret, f"{ts}.".encode() + body, hashlib.sha256)
    return f"t={ts},v1={mac.hexdigest()}"
 
TEMPLATES = {
  "servicenow": lambda e: {"short_description":
      f"[ARGOS] {e['severity']}: {e.get('challenge_id','')}",
      "description": json.dumps(e, ensure_ascii=False),
      "urgency": {"critica": 1, "alta": 2}.get(e.get("severity"), 3)},
  "jira": lambda e: {"fields": {"summary":
      f"[ARGOS] {e.get('challenge_id','evento')}",
      "description": json.dumps(e, ensure_ascii=False),
      "issuetype": {"name": "Bug"}}},
  "generic": lambda e: e,
}
 
@activity.defn(name="deliver_webhook_act")
async def deliver_webhook_act(delivery_id: int) -> str:
    with pool.connection() as db:
        d = db.execute("""SELECT w.url, w.secret, w.template, dl.event,
                dl.attempts FROM argos.webhook_deliveries dl
            JOIN argos.webhooks w ON w.id = dl.webhook_id
            WHERE dl.id=%s""", (delivery_id,)).fetchone()
    url, secret, template, event, attempts = d
    payload = TEMPLATES.get(template, TEMPLATES["generic"])(event)
    body = json.dumps(payload, ensure_ascii=False).encode()
    ts = str(int(time.time()))
    try:
        async with httpx.AsyncClient(timeout=15) as c:
            r = await c.post(url, content=body, headers={
                "Content-Type": "application/json",
                "X-Argos-Signature": _sign(bytes(secret), ts, body),
                "X-Argos-Event": event["type"]})
        ok = r.status_code < 300
        status = "entregado" if ok else f"http_{r.status_code}"
    except Exception as e:
        ok, status = False, f"error: {str(e)[:120]}"
    with pool.connection() as db:
        db.execute("""UPDATE argos.webhook_deliveries
            SET attempts=%s, status=%s, last_at=now() WHERE id=%s""",
            (attempts + 1, status, delivery_id))
        db.commit()
    if not ok and attempts + 1 < MAX_ATTEMPTS:
        raise RuntimeError(status)      # Temporal reintenta con backoff
    return status
 
# Suscriptor del bus: finding_opened / campaign_sealed / approval_requested
# crea filas en webhook_deliveries y lanza deliver_webhook_act por cada una.
